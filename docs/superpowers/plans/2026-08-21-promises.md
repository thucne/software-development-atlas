```ts
export type PromiseScenarioId =
  | 'return-value'
  | 'throw-error'
  | 'adopt-pending'
  | 'catch-recovery'
  | 'finally-transparent'
  | 'branching';

export type PromiseState = 'pending' | 'fulfilled' | 'rejected';

export type ResolutionKind =
  | 'unresolved'
  | 'fulfilled-value'
  | 'rejected-reason'
  | 'adopting';

export type PromiseNode = {
  id: string;
  label: string;
  state: PromiseState;
  resolution: ResolutionKind;
  value?: string;
  reason?: string;
  adopts?: string;
};

export type HandlerKind = 'then' | 'catch' | 'finally';

export type HandlerStep = {
  id: string;
  sourcePromiseId: string;
  resultPromiseId: string;
  kind: HandlerKind;
  label: string;
};

export type PromiseResolutionState = {
  scenarioId: PromiseScenarioId;
  promises: PromiseNode[];
  activeHandler: HandlerStep | null;
  output: string[];
  stepIndex: number;
  explanation: string;
  complete: boolean;
};

export type PromiseScenarioDefinition = {
  id: PromiseScenarioId;
  title: string;
  description: string;
  source: string;
};

export const PROMISE_SCENARIOS: readonly PromiseScenarioDefinition[];

export function createPromiseScenarioState(
  id: PromiseScenarioId,
): PromiseResolutionState;

export function stepPromiseScenario(
  state: PromiseResolutionState,
): PromiseResolutionState;
```

The model is a teaching model, not a Promise polyfill. It represents only the six approved predefined scenarios.

- [ ] **Step 1: Write the failing unit tests**

Create `tests/promise-resolution.test.ts` with tests equivalent to:

```ts
import {
  createPromiseScenarioState,
  stepPromiseScenario,
  type PromiseResolutionState,
  type PromiseScenarioId,
} from '@/lib/learning/promise-resolution';
import { describe, expect, it } from 'vitest';

function runScenario(id: PromiseScenarioId) {
  let state = createPromiseScenarioState(id);

  for (let guard = 0; guard < 50 && !state.complete; guard += 1) {
    state = stepPromiseScenario(state);
  }

  return state;
}

function findPromise(state: PromiseResolutionState, id: string) {
  const promise = state.promises.find((candidate) => candidate.id === id);
  if (!promise) throw new Error(`Missing promise ${id}`);
  return promise;
}

describe('promise resolution teaching model', () => {
  it('fulfills a downstream promise when a handler returns a plain value', () => {
    const state = runScenario('return-value');
    expect(findPromise(state, 'P1')).toMatchObject({
      state: 'fulfilled',
      resolution: 'fulfilled-value',
      value: '20',
    });
    expect(state.complete).toBe(true);
  });

  it('rejects a downstream promise when a handler throws', () => {
    const state = runScenario('throw-error');
    expect(findPromise(state, 'P1')).toMatchObject({
      state: 'rejected',
      resolution: 'rejected-reason',
      reason: 'boom',
    });
  });

  it('represents resolved-but-pending adoption before the adopted promise settles', () => {
    let state = createPromiseScenarioState('adopt-pending');
    let sawPendingAdoption = false;

    for (let guard = 0; guard < 50 && !state.complete; guard += 1) {
      state = stepPromiseScenario(state);
      const downstream = findPromise(state, 'P1');
      if (
        downstream.state === 'pending' &&
        downstream.resolution === 'adopting' &&
        downstream.adopts === 'P2'
      ) {
        sawPendingAdoption = true;
      }
    }

    expect(sawPendingAdoption).toBe(true);
    expect(findPromise(state, 'P1')).toMatchObject({
      state: 'fulfilled',
      resolution: 'fulfilled-value',
      value: '42',
    });
  });

  it('fulfills the promise returned by catch when the rejection handler recovers', () => {
    const state = runScenario('catch-recovery');
    expect(findPromise(state, 'P1')).toMatchObject({
      state: 'fulfilled',
      value: 'Guest',
    });
  });

  it('preserves the prior fulfillment through a successful finally handler', () => {
    const state = runScenario('finally-transparent');
    expect(findPromise(state, 'P1')).toMatchObject({
      state: 'fulfilled',
      value: '7',
    });
  });

  it('creates two independent downstream promises when then is called twice', () => {
    const state = runScenario('branching');
    expect(findPromise(state, 'P1')).toMatchObject({ state: 'fulfilled', value: '11' });
    expect(findPromise(state, 'P2')).toMatchObject({ state: 'fulfilled', value: '20' });
    expect(findPromise(state, 'P1').id).not.toBe(findPromise(state, 'P2').id);
  });
});
```

- [ ] **Step 3: Implement the minimal pure model**

Create `lib/learning/promise-resolution.ts` with immutable transitions and these scenario semantics:

### `return-value`

```text
P0 fulfilled with 10
→ create P1 pending
→ run then(value => value * 2)
→ P1 fulfilled with 20
→ complete
```

### `throw-error`

```text
P0 fulfilled with "start"
→ create P1 pending
→ run then(() => { throw new Error("boom") })
→ P1 rejected with "boom"
→ complete
```

### `adopt-pending`

```text
P0 fulfilled with "source"
→ create P1 pending and P2 pending
→ run then(() => P2)
→ P1 remains pending but resolution becomes adopting, adopts="P2"
→ P2 fulfills with 42
→ P1 fulfills with 42 after adopted outcome settles
→ complete
```

The step where P1 is `{ state: 'pending', resolution: 'adopting', adopts: 'P2' }` is normative.

### `catch-recovery`

```text
P0 rejected with "network"
→ create P1 pending
→ run catch(() => "Guest")
→ P1 fulfilled with "Guest"
→ complete
```

### `finally-transparent`

```text
P0 fulfilled with 7
→ create P1 pending
→ run finally(() => cleanup succeeds)
→ P1 fulfilled with original value 7
→ complete
```

Do not model the successful cleanup return value as replacing `7`.

### `branching`

```text
P0 fulfilled with 10
→ create distinct P1 and P2
→ run branch A handler: P1 fulfilled with 11
→ run branch B handler: P2 fulfilled with 20
→ complete
```

The explanation must explicitly state that the model's display order is illustrative and does not make one branch depend on the other.

`PROMISE_SCENARIOS` must include readable source snippets for all six scenarios.

- [ ] **Step 3: Include the normative raw-Markdown reasoning table**

The lesson must contain this table or semantically equivalent wording:

```md
| Handler result | Downstream Promise behavior |
| --- | --- |
| returns a plain value | fulfills with that value |
| returns normally with no value | fulfills with `undefined` |
| throws | rejects with the thrown reason |
| returns a fulfilled Promise/thenable | adopts and eventually fulfills with that outcome |
```

- [ ] **Step 4: Include the required chain examples**

- [ ] **Step 5: Include `finally()` semantics precisely**

Raw Markdown must say:

- successful cleanup preserves the original fulfillment value/rejection reason;
- throwing from `finally` rejects downstream with the thrown reason;
- returning a rejected promise/thenable rejects downstream with that adopted rejection;
- returning a still-pending promise/thenable can delay downstream propagation until cleanup settles;
- `finally(onFinally)` is not equivalent to `then(onFinally, onFinally)`.

- [ ] **Step 6: Include the combinator matrix and empty-input behavior**

Required matrix:

```md
| Need | API | Key failure behavior |
| --- | --- | --- |
| all inputs must fulfill | `Promise.all()` | rejects when an input rejects |
| observe every input outcome | `Promise.allSettled()` | fulfills with per-input result records |
| first fulfillment wins | `Promise.any()` | rejects with `AggregateError` if all reject |
| first settlement wins | `Promise.race()` | settles with the first settled input |
```

Required edge cases:

```text
Promise.all([]) -> fulfills with []
Promise.allSettled([]) -> fulfills with []
Promise.any([]) -> rejects with AggregateError
Promise.race([]) -> remains pending
```

Re-state that `Promise.all()` rejection does not cancel already-started operations.

- [ ] **Step 7: Include modern 2026 APIs**

`Promise.withResolvers()` example:

```js
const { promise, resolve, reject } = Promise.withResolvers();
```

Explain lifecycle ownership and that this is specialized, not a default construction style.

`Promise.try()` example:

```js
const result = Promise.try(fn, arg1, arg2);
```

State explicitly:

```text
Promise.try(fn) invokes fn synchronously.
Promise.resolve().then(fn) invokes fn later as Promise reaction work.
They are therefore not timing-equivalent.
```

Required raw rule:

```text
A Promise models an eventual result; it does not inherently own or cancel the underlying operation.
```

Mention `AbortSignal` only as a common Web API cancellation mechanism; do not implement cancellation in the lab.
